import { config } from "dotenv"; config(); import express from "express"; import { createServer } from "http"; import { registerRoutes } from "../server/routes"; const app = express(); app.use(express.json({ limit: "50mb" })); app.use(express.urlencoded({ extended: false })); const server = createServer(app); const ready = registerRoutes(server, app); export default async function handler(req: any, res: any) { await ready; app(req, res); }
